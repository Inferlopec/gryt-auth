# Caddy — blocks for /opt/caddy/Caddyfile (same-host layout: Caddy and the
# identity stack on the same machine, Caddy with host networking).
# Insert INSIDE the existing Caddyfile (at the top-level site block level),
# after taking a backup: cp /opt/caddy/Caddyfile /opt/caddy/Caddyfile.bak-$(date +%F)
# Then: docker exec inferlo-caddy caddy reload --config /etc/caddy/Caddyfile
#
# The identity stack listens on 127.0.0.1 only (see docker-compose.yml),
# so these services are not reachable externally even without these rules.
# This file is a snippet for the operator to apply; it is not applied automatically.

    reverse_proxy 127.0.0.1:18080
}

id.inferlo.xyz {
    reverse_proxy 127.0.0.1:18081
}
